-- ====================== HELPERS ======================
local function getPing()
    local p = 80
    pcall(function()
        local item = Stats.Network.ServerStatsItem["Data Ping"]
        if item then
            p = tonumber((item:GetValueString() or ""):match("%d+")) or 80
        end
    end)
    return p
end

local function getRoot()
    local c = LP.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end

local function inRound()
    local c = LP.Character
    if not c then return false end
    local p = workspace:FindFirstChild("Players")
    local s = p and p:FindFirstChild("Survivors")
    return s and c:IsDescendantOf(s)
end

local function getKillers()
    local p = workspace:FindFirstChild("Players")
    return p and p:FindFirstChild("Killers")
end

local function behead(killer)
    if not S.IgnoreBehead or not killer then return false end
    local hum = killer:FindFirstChildOfClass("Humanoid")
    local anim = hum and hum:FindFirstChildOfClass("Animator")
    if not anim then return false end
    for _, track in ipairs(anim:GetPlayingAnimationTracks()) do
        local n = ((track.Animation and track.Animation.Name) or ""):lower()
        if n:find("behead") or n:find("gashing") or n:find("execute") then
            return true
        end
    end
    return false
end

local function isThreatening(kr)
    local my = getRoot()
    if not my or not kr then return false end

    local toMe = (my.Position - kr.Position)
    if toMe.Magnitude < 0.3 then return true end
    local dirToMe = toMe.Unit

    if S.FaceCheck or S.AntiBait then
        if kr.CFrame.LookVector:Dot(dirToMe) < S.Facing then
            return false
        end
    end

    if S.AntiBait then
        local vel = kr.AssemblyLinearVelocity
        local horiz = Vector3.new(vel.X, 0, vel.Z)
        if horiz.Magnitude > 1.8 then
            if horiz.Unit:Dot(dirToMe) < S.ApproachDot then
                return false
            end
        end
    end
    return true
end

-- ====================== FIRE ======================
local function fireBlock()
    if tick() - lastBlock < 0.22 then return end
    lastBlock = tick()

    if Remote then
        pcall(function()
            Remote:FireServer("UseActorAbility", {"Block"})
        end)
    end

    if not isMobile then
        pcall(function()
            VIM:SendKeyEvent(true, Enum.KeyCode.Q, false, game)
            task.wait(0.015)
            VIM:SendKeyEvent(false, Enum.KeyCode.Q, false, game)
        end)
    end
end

local function firePunch()
    if Remote then
        pcall(function()
            Remote:FireServer("UseActorAbility", {"Punch"})
        end)
    end
    if not isMobile then
        pcall(function()
            VIM:SendKeyEvent(true, Enum.KeyCode.R, false, game)
            task.wait(0.015)
            VIM:SendKeyEvent(false, Enum.KeyCode.R, false, game)
        end)
    end
end

-- ====================== DETECTION ======================
local function isM1(sound)
    local n = (sound.Name or ""):lower()
    return (n:find("swing") or n:find("slash") or n:find("attack") or n:find("m1") or n:find("claw") or n:find("tag")) 
        and not (n:find("foot") or n:find("step") or n:find("walk") or n:find("run"))
end

local function junk(sound)
    local n = (sound.Name or ""):lower()
    if n:find("foot") or n:find("step") or n:find("walk") or n:find("run") then return true end
    if sound.Looped or (sound.TimeLength and sound.TimeLength > 2.3) then return true end
    return false
end

local function tryBlock(kr, killer)
    if not S.AutoBlock or not inRound() then return end
    if tick() - lastTrigger < 0.08 then return end

    local my = getRoot()
    if not my or not kr then return end

    local dist = (my.Position - kr.Position).Magnitude
    if dist > S.Range + (S.HighPing and 5 or 0) then return end
    if behead(killer) then return end
    if not isThreatening(kr) then return end

    lastTrigger = tick()

    local delay = S.Delay
    local ping = getPing()
    if S.HighPing then
        if ping >= 180 then delay = delay + 0.035
        elseif ping >= 120 then delay = delay + 0.018 end
    end
    if S.CloseRangeBoost and dist < 10 then
        delay = math.max(0, delay - 0.012)
    end

    task.delay(delay, function()
        if not S.AutoBlock then return end
        fireBlock()

        if S.AutoPunch then
            task.delay(0.16, function()
                if S.PunchAim then aimUntil = tick() + 0.45 end
                firePunch()
            end)
        end

        if S.HitboxDrag then
            hdtTarget = kr
            hdtUntil = tick() + S.HDTTime
        end
    end)
end

local function onSound(sound, killer)
    if not S.AutoBlock or junk(sound) then return end
    local kr = killer:FindFirstChild("HumanoidRootPart")
    if not kr then return end

    if isM1(sound) or (sound.TimeLength and sound.TimeLength > 0.03 and sound.TimeLength < 1.8) then
        tryBlock(kr, killer)
    end
end

local function hook(killer)
    if hooked[killer] then return end
    hooked[killer] = true

    for _, obj in ipairs(killer:GetDescendants()) do
        if obj:IsA("Sound") then
            obj.Played:Connect(function() onSound(obj, killer) end)
        end
    end

    killer.DescendantAdded:Connect(function(obj)
        if obj:IsA("Sound") then
            obj.Played:Connect(function() onSound(obj, killer) end)
        end
    end)

    local hum = killer:FindFirstChildOfClass("Humanoid")
    if hum then
        hum.AnimationPlayed:Connect(function(track)
            if not S.AutoBlock then return end
            local n = ((track.Animation and track.Animation.Name) or ""):lower()
            if (n:find("slash") or n:find("swing") or n:find("attack") or n:find("m1") or n:find("claw")) 
                and not (n:find("behead") or n:find("idle") or n:find("walk")) then
                local kr = killer:FindFirstChild("HumanoidRootPart")
                if kr then tryBlock(kr, killer) end
            end
        end)
    end
end

-- ====================== LOOPS ======================
task.spawn(function()
    while true do
        task.wait(0.45)
        local ks = getKillers()
        if ks then
            for _, k in ipairs(ks:GetChildren()) do
                hook(k)
            end
        end
        for k in pairs(hooked) do
            if not k.Parent then hooked[k] = nil end
        end
    end
end)

task.spawn(function()
    while true do
        task.wait(0.11)
        if not S.AutoBlock or not inRound() then continue end
        if tick() - lastTrigger < 0.09 then continue end

        local my = getRoot()
        if not my then continue end
        local ks = getKillers()
        if not ks then continue end

        for _, k in ipairs(ks:GetChildren()) do
            local kr = k:FindFirstChild("HumanoidRootPart")
            local hum = k:FindFirstChildOfClass("Humanoid")
            if not kr or not hum then continue end
            if (my.Position - kr.Position).Magnitude > S.Range + 2 then continue end
            if behead(k) then continue end

            local anim = hum:FindFirstChildOfClass("Animator")
            if not anim then continue end

            for _, track in ipairs(anim:GetPlayingAnimationTracks()) do
                local n = ((track.Animation and track.Animation.Name) or ""):lower()
                if (n:find("slash") or n:find("swing") or n:find("attack") or n:find("m1")) and track.TimePosition < 0.45 then
                    tryBlock(kr, k)
                    break
                end
            end
        end
    end
end)

task.spawn(function()
    while true do
        task.wait(0.06)
        if S.InfStamina and inRound() then
            pcall(function()
                local mod = require(ReplicatedStorage.Systems.Character.Game.Sprinting)
                if mod and typeof(mod.Stamina) == "number" then
                    mod.Stamina = 1.4
                end
            end)
        end
    end
end)

RunService.RenderStepped:Connect(function()
    if not (S.PunchAim and tick() < aimUntil) then return end
    local my = getRoot()
    if not my then return end
    local ks = getKillers()
    if not ks then return end

    local best, bd = nil, 999
    for _, k in ipairs(ks:GetChildren()) do
        local r = k:FindFirstChild("HumanoidRootPart")
        if r then
            local d = (my.Position - r.Position).Magnitude
            if d < bd then best, bd = r, d end
        end
    end
    if best then
        local look = (best.Position - my.Position) * Vector3.new(1,0,1)
        if look.Magnitude > 0.05 then
            my.CFrame = my.CFrame:Lerp(CFrame.lookAt(my.Position, my.Position + look), 0.4)
        end
    end
end)

RunService.Heartbeat:Connect(function()
    if not S.HitboxDrag or not hdtTarget or not hdtTarget.Parent or tick() > hdtUntil then return end
    local my = getRoot()
    if not my or not inRound() then return end

    local targetPos = hdtTarget.Position
    local myPos = my.Position
    local dist = (targetPos - myPos).Magnitude
    if dist < 4.5 then return end

    local flatDir = Vector3.new(targetPos.X - myPos.X, 0, targetPos.Z - myPos.Z)
    if flatDir.Magnitude < 0.1 then return end
    flatDir = flatDir.Unit

    local speed = S.HDTSpeed
    if dist < 9 then speed = speed * 0.75 end

    local vel = my.AssemblyLinearVelocity
    my.AssemblyLinearVelocity = vel:Lerp(Vector3.new(flatDir.X * speed, vel.Y, flatDir.Z * speed), 0.28)
    my.CFrame = my.CFrame:Lerp(CFrame.lookAt(myPos, Vector3.new(targetPos.X, myPos.Y, targetPos.Z)), 0.15)
end)
